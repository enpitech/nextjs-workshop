# Exercise 9, Task 4, the wrong way: cookies() inside a "use cache" function. The Greeting that calls it is inside <Suspense>.
# Captured 2026-10-07 with Next.js 16.3.5, cacheComponents: true. The <Suspense> does not help: the build still fails.

## board/data.ts (the function that breaks the rule)

export async function getMe() {
  "use cache";
  cacheLife("days");
  const id = (await cookies()).get("speaker")?.value;
  return id ? fetchSpeaker(Number(id)) : null;
}

## board/greeting.tsx

import { getMe } from "./data";

export async function Greeting() {
  const me = await getMe();
  return <p>Signed in as {me ? me.name : "a guest"}.</p>;
}

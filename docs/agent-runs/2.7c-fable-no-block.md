+// /speakers lists every speaker. The list is read on every request, never at build time.
+// connection() marks the section dynamic. The Suspense boundary lets the shell prerender
+// while the list streams in, which cacheComponents requires for uncached reads.
+import { Suspense } from "react";
+import { connection } from "next/server";
+import { read } from "@/lib/db";
+import type { Speaker } from "@/lib/types";
+import { BackButton } from "./back-button";
+
+export default function Speakers() {
+  return (
+    <main>
+      <h1 className="mb-4 text-2xl font-bold">Speakers</h1>
+      <BackButton />
+      <Suspense fallback={<p className="text-sm text-gray-500">Loading speakers…</p>}>
+        <SpeakersList />
+      </Suspense>
+    </main>
+  );
+}
+
+async function SpeakersList() {
+  await connection();
+  const speakers = await read<Speaker[]>("/speakers");
+  return (
+    <ul className="my-4 space-y-3">
+      {speakers.map((s) => (
+        <li key={s.id}>
+          <p className="font-semibold">{s.name}</p>
+          <p className="text-sm text-gray-500">{s.bio}</p>
+        </li>
+      ))}
+    </ul>
+  );

import { defineTool } from "eve/tools";
import { z } from "zod";
import { getTalk } from "@/lib/talks";

// Reuse the cached read from the app. It is fast, and it is already tested.
export default defineTool({
  description: "Get one conference talk by its id, with its vote count.",
  inputSchema: z.object({ id: z.number() }),
  async execute({ id }) {
    const talk = await getTalk(id);
    return { id: talk.id, title: talk.title, votes: talk.votes };
  },
});

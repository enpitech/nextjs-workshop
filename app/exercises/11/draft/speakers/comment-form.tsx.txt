import { addComment } from "./actions";

// A form with a Server Action. The hidden field sends the talk id.
export function CommentForm({ talkId }: { talkId: number }) {
  return (
    <form action={addComment} className="flex gap-2">
      <input type="hidden" name="talkId" value={talkId} />
      <input name="text" required maxLength={280} className="flex-1 rounded border px-2 py-1" />
      <button className="rounded border px-3 py-1">Add comment</button>
    </form>
  );
}
